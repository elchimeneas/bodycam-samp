cmake_minimum_required(VERSION 3.21)
project(Bodycam VERSION 0.4.2 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Bodycam requires Windows x86 (-A Win32)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(bodycam_core STATIC src/config.cpp src/renderer.cpp src/body_camera.cpp)
target_include_directories(bodycam_core PUBLIC src)
target_compile_definitions(bodycam_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(bodycam_core PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(bodycam_core PUBLIC d3d9 gdi32 user32)
add_library(Bodycam SHARED src/plugin.cpp resources/bodycam.rc)
target_link_libraries(Bodycam PRIVATE bodycam_core)
target_compile_options(Bodycam PRIVATE /W4 /utf-8 /permissive-)
target_link_options(Bodycam PRIVATE "/PDBALTPATH:%_PDB%")
set_target_properties(Bodycam PROPERTIES SUFFIX ".asi" PREFIX "")
add_executable(bodycam_test tests/tests.cpp)
target_link_libraries(bodycam_test PRIVATE bodycam_core windowscodecs ole32)
add_dependencies(bodycam_test Bodycam)
target_compile_definitions(bodycam_test PRIVATE BODYCAM_TEST_BINARY="$<TARGET_FILE:Bodycam>" BODYCAM_TEST_INI="${CMAKE_CURRENT_SOURCE_DIR}/Bodycam.ini")
target_compile_options(bodycam_test PRIVATE /W4 /utf-8 /permissive-)
enable_testing()
add_test(NAME bodycam_native_validation COMMAND bodycam_test)
add_executable(bodycam_camera_test tests/camera_tests.cpp)
target_link_libraries(bodycam_camera_test PRIVATE bodycam_core)
target_compile_options(bodycam_camera_test PRIVATE /W4 /utf-8 /permissive-)
add_test(NAME bodycam_camera_validation COMMAND bodycam_camera_test)
